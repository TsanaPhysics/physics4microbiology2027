\begin{planbox}[แผนบริหารการสอนประจำบทที่ 1\quad ระบบหน่วยและการวัดในโลกจุลทรรศน์ (สัปดาห์ที่ 1)]
\noindent\textbf{1. หัวข้อเนื้อหาประจำบท (บรรยาย 2 ชั่วโมง และปฏิบัติการ 3 ชั่วโมง)}
\begin{itemize}
    \item 1.1 ความสำคัญของระบบหน่วยเอสไอและคำอุปสรรคในงานจุลชีววิทยา
    \item 1.2 การแปลงหน่วยขนาดเซลล์จุลินทรีย์และปริมาตรสารละลายโดยใช้เทคนิคบันไดเลื่อนจุดทศนิยม
    \item 1.3 เลขนัยสำคัญ ความเที่ยง ความแม่น และความคลาดเคลื่อนของการวัดในห้องปฏิบัติการ
    \item 1.4 ปฏิบัติการที่ 1 การวัดละเอียดด้วยเวอร์เนียร์ ไมโครมิเตอร์ และทักษะการใช้ไมโครปิเปตและเครื่องชั่งดิจิทัล
\end{itemize}

\noindent\textbf{2. วัตถุประสงค์เชิงพฤติกรรมและผลลัพธ์การเรียนรู้รายวิชา (CLOs ที่เชื่อมโยง PLOs)}
\begin{enumerate}
    \item อธิบายระบบหน่วยมาตรฐานสากลและคำอุปสรรคที่ใช้ระบุขนาดของแบคทีเรีย ยีสต์ ไวรัส และปริมาตรสารละลายได้ถูกต้อง \textbf{(CLO1 เชื่อมโยง PLO2)}
    \item คำนวณแปลงหน่วยระหว่างมิลลิเมตร ไมโครเมตร นาโนเมตร และระหว่างมิลลิลิตร ไมโครลิตร โดยไม่ต้องใช้คณิตศาสตร์ซับซ้อนได้ \textbf{(CLO1, CLO2 เชื่อมโยง PLO2, PLO3)}
    \item ปฏิบัติการใช้เวอร์เนียร์คาลิปเปอร์ ไมโครมิเตอร์ เครื่องชั่งละเอียด และไมโครปิเปตอย่างถูกวิธี มีวินัย และบันทึกผลตามความเป็นจริง \textbf{(CLO3 เชื่อมโยง PLO1)}
    \item ร่วมกันวิเคราะห์ความคลาดเคลื่อนของการวัดและเขียนรายงานปฏิบัติการกลุ่มได้อย่างถูกต้อง \textbf{(CLO4 เชื่อมโยง PLO5, PLO6)}
\end{enumerate}

\noindent\textbf{3. กิจกรรมการจัดการเรียนรู้เชิงรุก (Active Learning \& OBE)}
\begin{itemize}
    \item \textbf{ขั้นนำเข้าสู่บทเรียน}\quad ผู้สอนยกตัวอย่างปัญหาจริงในห้องแล็บจุลชีววิทยา เช่น การดูดสารละลายผิดหน่วยระหว่าง $100\,\mu\text{L}$ กับ $1\text{ mL}$ ซึ่งส่งผลต่อความเข้มข้นของเชื้อแบคทีเรียถึง 10 เท่า
    \item \textbf{ขั้นสอนและฝึกทักษะคำนวณแบบเข้าใจง่าย}\quad ใช้แผนภาพบันไดคำอุปสรรค ($\times 1{,}000$ และ $\div 1{,}000$) แทนการคิดเลขยกกำลังที่ซับซ้อน พร้อมทำใบงานคู่คิด (Think-Pair-Share)
    \item \textbf{ขั้นปฏิบัติการทดลอง}\quad นักศึกษาแบ่งกลุ่มฝึกวัดขนาดจานเพาะเชื้อ หลอดทดลอง และทดสอบความแม่นของการดูดน้ำกลั่นด้วยไมโครปิเปตบนเครื่องชั่งทศนิยม 4 ตำแหน่ง
\end{itemize}

\noindent\textbf{4. สื่อการเรียนรู้และแหล่งเรียนรู้}
\begin{itemize}
    \item เอกสารคำสอนบทที่ 1 และตารางบันไดคำอุปสรรคสำหรับนักจุลชีววิทยา
    \item ชุดเครื่องมือวัดละเอียด (Vernier Caliper, Micrometer), ไมโครปิเปตขนาด P20, P200, P1000 และเครื่องชั่งดิจิทัล
\end{itemize}

\noindent\textbf{5. การวัดและประเมินผลประจำบท}
\begin{itemize}
    \item แบบทดสอบก่อนเรียนและหลังเรียนเรื่องการแปลงหน่วยระดับไมโครและนาโน \textbf{(CLO1)}
    \item การสังเกตทักษะการจับไมโครปิเปต การอ่านสเกล และความซื่อสัตย์ในการบันทึกข้อมูล \textbf{(CLO3)}
    \item รายงานปฏิบัติการที่ 1 เรื่องการวัดและเครื่องมือพื้นฐานในงานจุลชีววิทยา \textbf{(CLO4)}
\end{itemize}
\end{planbox}

\section{ระบบหน่วยสากลเอสไอ (SI Units) วิวัฒนาการจากอดีตสู่ปัจจุบัน และบทบาทในงานจุลชีววิทยา}

\deepskinfig{figures/sec_3d/ch01_scale_3d.jpg}{การเปรียบเทียบสเกลขนาดในโลกจุลชีววิทยา 3 มิติ: ตั้งแต่ระดับโคโลนีบนจานเพาะเชื้อ ($\text{mm}$), เซลล์แบคทีเรียเดี่ยว ($\mu\text{m}$) ไปจนถึงอนุภาคแบคเทอริโอเฟจและชีวโมเลกุลเกลียวคู่ดีเอ็นเอ ($\text{nm}$) พร้อมเครื่องมือวัดละเอียดมาตรฐานสากล}

ระบบหน่วยระหว่างชาติ หรือ \textbf{ระบบหน่วยเอสไอ} (\textbf{International System of Units} หรือ \textbf{Le Système International d'Unités --- SI}) เป็นภาษาสากลทางวิทยาศาสตร์ที่ทำให้นักจุลชีววิทยาทั่วโลกสามารถสื่อสารข้อมูลเชิงปริมาณร่วมกันได้อย่างเที่ยงตรง ไม่ว่าจะเป็นขนาดของเซลล์ ความหนาของเยื่อหุ้มเซลล์ อุณหภูมิการบ่มเพาะเชื้อ เวลาการแบ่งเซลล์ ปริมาตรสารละลายบัฟเฟอร์ และความเข้มข้นของสารอาหาร

\begin{conceptbox}[การปฏิวัติประวัติศาสตร์ระบบหน่วยเอสไอ ค.ศ. 2019 (The 2019 SI Redefinition)]
ในอดีต ระบบหน่วยวัดผูกโยงอยู่กับ \textbf{``วัตถุประดิษฐ์ทางกายภาพ (Physical Artifacts)''} เช่น แท่งเมตรมาตรฐานแพลทินัม-อิริเดียม หรือก้อนกิโลกรัมต้นแบบนานาชาติ (\textit{Le Grand K}) ซึ่งมีข้อจำกัดร้ายแรงคือ \textit{มวลสามารถเปลี่ยนแปลงได้ตามกาลเวลาจากการสึกหรอหรือปนเปื้อนฝุ่นผงระดับไมโครกรัม} และหากวัตถุต้นแบบถูกทำลาย มาตรฐานการวัดของมนุษยชาติจะสูญสิ้นไปทันที

จนกระทั่งเมื่อวันที่ \textbf{20 พฤษภาคม ค.ศ. 2019} ที่ประชุมใหญ่ว่าด้วยการชั่งตวงวัด (CGPM / BIPM) ได้ประกาศการเปลี่ยนแปลงครั้งประวัติศาสตร์ โดย \textbf{``ยกเลิกการนิยามหน่วยผ่านวัตถุประดิษฐ์ทั้งหมด''} แล้วเปลี่ยนมากำหนดค่าคงที่ที่แน่นอน (Exact Numerical Values โดยไม่มีความคลาดเคลื่อน $\Delta = 0$) ให้กับ \textbf{7 ค่าคงตัวพื้นฐานแห่งเอกภพ (The 7 Defining Physical Constants)} ทำให้ทุกห้องปฏิบัติการทั่วโลกสามารถสร้างมาตรฐานหน่วยวัดที่เที่ยงตรงได้ด้วยตนเอง
\end{conceptbox}

\deepskinfig{figures/sec_3d/ch01_artifacts_pixar_3d.jpg}{แบบจำลอง 3 มิติวัตถุประดิษฐ์ต้นแบบทางกายภาพในอดีต (Historical Physical Artifacts in 3D Pixar): แท่งเมตรมาตรฐานหน้าตัด Tresca (1889), ก้อนมวลกิโลกรัม Le Grand K (1889) ในครอบแก้วสุญญากาศ 3 ชั้น, และหอนาฬิกาอะตอมซีเซียม ณ ห้องนิรภัยมาตรวิทยา BIPM ก่อนการปฏิวัติสู่ 7 ค่าคงตัวพื้นฐาน ค.ศ. 2019}

\deepskinfig{figures/sec_3d/ch01_si_wheel_3d.jpg}{แบบจำลอง 3 มิติสไตล์พิกซาร์ วงล้อ 7 หน่วยฐานสากลเอสไอ (The 7 SI Base Units 3D Pixar Wheel): โครงข่าย 7 หน่วยฐานหลัก ได้แก่ เมตร ($\text{m}$), กิโลกรัม ($\text{kg}$), วินาที ($\text{s}$), แอมแปร์ ($\text{A}$), เคลวิน ($\text{K}$), โมล ($\text{mol}$) และแคนเดลา ($\text{cd}$) ถ่ายทอดผ่านมิติแสงเงา 3 มิติสมจริง พร้อมไอคอนเครื่องมือวัดวิทยาศาสตร์และคำอธิบาย 3 ภาษา (ไทย, อังกฤษ, จีน)}

\subsection{เจาะลึก 7 หน่วยฐานเอสไอ (7 SI Base Units): นิยามจากอดีตสู่ปัจจุบัน}

หน่วยฐานเอสไอประกอบด้วย 7 ปริมาณหลักที่อิสระต่อกันในเชิงมิติ ได้แก่:

\begin{enumerate}
    \item \textbf{ความยาว (Length, $l$) --- เมตร (Metre, $\text{m}$):}
    \begin{itemize}
        \item \textit{อดีต (1791):} $\tfrac{1}{10{,}000{,}000}$ ของระยะทางจากขั้วโลกเหนือถึงเส้นศูนย์สูตรผ่านปารีส ต่อมา (1889) ใช้ระยะระหว่างรอยขีดบนแท่งโลหะผสม $\text{Pt-Ir}$ และ (1960) ใช้ $1{,}650{,}763.73$ เท่าของความยาวคลื่นแสงสีส้ม-แดงของอะตอม $^{86}\text{Kr}$
        \item \textit{ปัจจุบัน (ตั้งแต่ 1983):} \textbf{ระยะทางที่แสงเคลื่อนที่ได้ในสุญญากาศ ในช่วงเวลา $\frac{1}{299{,}792{,}458}$ วินาที} โดยตรึงอัตราเร็วแสงในสุญญากาศ $c = 299{,}792{,}458\text{ m/s}$ อย่างแน่นอน
        \item \textit{บทบาทในจุลชีววิทยา:} วัดขนาดเซลล์แบคทีเรีย ($\mu\text{m}$), ขนาดอนุภาคไวรัส ($\text{nm}$), และความยาวคลื่นแสงส่องเซลล์ ($\text{OD}_{600}$)
    \end{itemize}
    \deepskinfig{figures/sec_3d/ch01_metre_tresca_3d.jpg}{แบบจำลอง 3 มิติแท่งเมตรมาตรฐาน $\text{Pt-Ir}$ หน้าตัดกากบาทแบบเทรสกา (Tresca X-section) บนลูกกลิ้งจุดแอรี่ในกล่องไม้มะฮอกกานี พร้อมลำแสงเลเซอร์แสดงนิยามใหม่ผ่านความเร็วแสง $c = 299{,}792{,}458\text{ m/s}$}

    \item \textbf{มวล (Mass, $m$) --- กิโลกรัม (Kilogram, $\text{kg}$):}
    \begin{itemize}
        \item \textit{อดีต (1795--2019):} มวลของน้ำบริสุทธิ์ $1\text{ L}$ ที่ $4\,^\circ\text{C}$ ต่อมา (1889) หล่อเป็นก้อนทรงกระบอกโลหะผสม $\text{Pt-Ir}$ ต้นแบบนานาชาติ (\textit{Le Grand K}) เก็บรักษาในครอบแก้วสุญญากาศ 3 ชั้น ณ BIPM ประเทศฝรั่งเศส (มวลลอยตัวเปลี่ยนไป $\sim 50\,\mu\text{g}$ ในรอบ 100 ปี)
        \item \textit{ปัจจุบัน (ตั้งแต่ 20 พฤษภาคม 2019):} \textbf{นิยามผ่านค่าคงตัวของพลังค์ ($h$) ให้มีค่าเท่ากับ $6.626\,070\,15 \times 10^{-34}\text{ kg}\cdot\text{m}^2\cdot\text{s}^{-1}$ อย่างแน่นอน} โดยชั่งผ่านเครื่องชั่งคิบบิล (Kibble Balance)
        \item \textit{บทบาทในจุลชีววิทยา:} การชั่งสารเคมีเตรียมอาหารเลี้ยงเชื้อ ($\text{g, mg}$), การชั่งน้ำกลั่นสอบเทียบไมโครปิเปต ($1{,}000\,\mu\text{L} = 1.0000\text{ g}$)
    \end{itemize}
    \deepskinfig{figures/sec_3d/ch01_ipk_grand_k_3d.jpg}{ก้อนกิโลกรัมต้นแบบนานาชาติ (Le Grand K / IPK) ทรงกระบอกโลหะผสม $\text{Pt-Ir}$ ภายใต้ครอบแก้วสุญญากาศ 3 ชั้น ณ BIPM ประเทศฝรั่งเศส}
    \deepskinfig{figures/sec_3d/ch01_kibble_balance_3d.jpg}{แบบจำลอง 3 มิติเครื่องชั่งคิบบิล (Kibble Balance) สร้างสมดุลแรงโน้มถ่วงทางกล ($mg$) กับแรงแม่เหล็กไฟฟ้า ($BLI$) ในระดับควอนตัม สำหรับนิยาม 1 กิโลกรัมใหม่ผ่านค่าคงตัวของพลังค์ ($h$)}

    \item \textbf{เวลา (Time, $t$) --- วินาที (Second, $\text{s}$):}
    \begin{itemize}
        \item \textit{อดีต:} $\frac{1}{86{,}400}$ ของวันสุริยคติเฉลี่ย (Mean Solar Day) แต่พบว่าโลกหมุนช้าลงจากแรงเสียดทานน้ำขึ้นน้ำลง
        \item \textit{ปัจจุบัน (ตั้งแต่ 1967):} \textbf{ช่วงเวลาเท่ากับ $9{,}192{,}631{,}770$ คาบของการแผ่รังสี} จากการเปลี่ยนระดับพลังงานไฮเปอร์ไฟน์ในสถานะพื้นของอะตอมซีเซียม-133 ($^{133}\text{Cs}$) ที่ศูนย์สัมบูรณ์
        \item \textit{บทบาทในจุลชีววิทยา:} เวลาการแบ่งเซลล์ (Generation Time), เวลานึ่งฆ่าเชื้อ Autoclave ($15\text{--}20\text{ min} = 900\text{--}1{,}200\text{ s}$)
    \end{itemize}
    \deepskinfig{figures/sec_3d/ch01_def_second_3d.jpg}{แบบจำลอง 3 มิติไทม์ไลน์วิวัฒนาการนิยาม 1 วินาที: จากการหมุนรอบตัวเองของโลก (วันสุริยคติเฉลี่ย) สู่วงโคจรโลก (ปีทรอปิก) จนถึงการสั่นไฮเปอร์ไฟน์ของอะตอมซีเซียม-133}
    \deepskinfig{figures/sec_3d/ch01_cesium_clock_3d.jpg}{หอนาฬิกาอะตอมน้ำพุซีเซียม-133 (Cesium-133 Fountain Clock NIST-F2) ลำแสงเลเซอร์ทำให้อะตอมเย็นตัวยิ่งยวด (Laser cooling) ยิงผ่านโพรงคลื่นไมโครเวฟแรมซีย์ ตรึงความถี่ $\Delta\nu_{\text{Cs}} = 9{,}192{,}631{,}770\text{ Hz}$}

    \item \textbf{กระแสไฟฟ้า (Electric current, $I$) --- แอมแปร์ (Ampere, $\text{A}$):}
    \begin{itemize}
        \item \textit{ปัจจุบัน (2019):} ตรึงค่าประจุไฟฟ้ามูลฐาน $e = 1.602\,176\,634 \times 10^{-19}\text{ C}$ ($\text{A}\cdot\text{s}$)
        \item \textit{บทบาทในจุลชีววิทยา:} กระแสไฟฟ้าในเจลอิเล็กโทรโฟเรซิสแยกสาย DNA และโปรตีน, ศักย์ไฟฟ้าเยื่อหุ้มเซลล์
    \end{itemize}

    \item \textbf{อุณหภูมิอุณหพลวัต (Thermodynamic temperature, $T$) --- เคลวิน (Kelvin, $\text{K}$):}
    \begin{itemize}
        \item \textit{ปัจจุบัน (2019):} ตรึงค่าคงตัวบ็อลทซ์มัน $k = 1.380\,649 \times 10^{-23}\text{ J/K}$ โดยสัมพันธ์กับเซลเซียสผ่านสมการ $T_{\text{K}} = T_{^\circ\text{C}} + 273.15$
        \item \textit{บทบาทในจุลชีววิทยา:} ตู้บ่มเชื้อ ($37\,^\circ\text{C} = 310.15\text{ K}$), Autoclave ($121\,^\circ\text{C} = 394.15\text{ K}$), PCR Denaturation ($95\,^\circ\text{C}$)
    \end{itemize}

    \item \textbf{ปริมาณของสาร (Amount of substance, $n$) --- โมล (Mole, $\text{mol}$):}
    \begin{itemize}
        \item \textit{ปัจจุบัน (2019):} มีจำนวนอนุภาคมูลฐานเท่ากับ $6.022\,140\,76 \times 10^{23}$ อนุภาค ซึ่งคือค่าคงตัวอาโวกาโดร ($N_{\text{A}}$)
        \item \textit{บทบาทในจุลชีววิทยา:} การเตรียมสารละลายความเข้มข้นโมลาร์ ($\text{M} = \text{mol/L}$), ปริมาณสารตั้งต้นในปฏิกิริยาชีวเคมี
    \end{itemize}

    \item \textbf{ความเข้มของการส่องสว่าง (Luminous intensity, $I_v$) --- แคนเดลา (Candela, $\text{cd}$):}
    \begin{itemize}
        \item \textit{ปัจจุบัน:} ตรึงประสิทธิผลการส่องสว่าง $K_{\text{cd}} = 683\text{ lm/W}$ ของรังสีความถี่ $540 \times 10^{12}\text{ Hz}$
        \item \textit{บทบาทในจุลชีววิทยา:} การให้แสงสว่างในการเพาะเลี้ยงสาหร่ายเซลล์เดียวและไซยาโนแบคทีเรีย, การเรืองแสงชีวภาพ (Bioluminescence)
    \end{itemize}
\end{enumerate}

\begin{itemize}
    \item \textbf{แปลงจากหน่วยใหญ่ไปหน่วยเล็ก} (เดินลงบันได 1 ขั้น)\quad \textbf{ให้คูณด้วย $1{,}000$} (หรือเลื่อนจุดทศนิยมไปทางขวา 3 ตำแหน่ง) เช่น $1\text{ mL} = 1{,}000\,\mu\text{L}$ และ $1\,\mu\text{m} = 1{,}000\text{ nm}$
    \item \textbf{แปลงจากหน่วยเล็กไปหน่วยใหญ่} (เดินขึ้นบันได 1 ขั้น)\quad \textbf{ให้หารด้วย $1{,}000$} (หรือเลื่อนจุดทศนิยมไปทางซ้าย 3 ตำแหน่ง) เช่น $250\,\mu\text{L} = 0.250\text{ mL}$
\end{itemize}
\end{conceptbox}

\begin{worked}[การแปลงหน่วยขนาดเซลล์แบคทีเรียและปริมาตรไมโครปิเปต]
\textbf{โจทย์ปัญหา}\quad นักศึกษาจุลชีววิทยาส่องกล้องจุลทรรศน์พบเซลล์แบคทีเรีย \textit{Bacillus subtilis} มีความยาวเฉลี่ย $4.5\,\mu\text{m}$ และในการเตรียมปฏิกิริยาพีซีอาร์ (PCR) ต้องใช้ไมโครปิเปตดูดสารละลายบัฟเฟอร์ปริมาตร $0.05\text{ mL}$ จงหาว่า (ก) แบคทีเรียชนิดนี้มีความยาวกี่มิลลิเมตร ($\text{mm}$) และกี่นาโนเมตร ($\text{nm}$) และ (ข) ต้องตั้งค่าไมโครปิเปตเป็นกี่ไมโครลิตร ($\mu\text{L}$)

\textbf{วิธีทำทีละขั้นตอนฉบับเข้าใจง่าย}
\begin{enumerate}
    \item \textbf{ขั้นที่ 1 แปลงความยาวแบคทีเรีย $4.5\,\mu\text{m}$ เป็น $\text{mm}$ และ $\text{nm}$}
    \begin{itemize}
        \item จาก $\mu\text{m}$ ไปเป็น $\text{mm}$ คือการเดินขึ้นบันได 1 ขั้น (หน่วยเล็กไปหน่วยใหญ่) จึงนำมา\textbf{หารด้วย $1{,}000$}
        \[
            \text{ความยาวในหน่วยมิลลิเมตร} = \frac{4.5}{1{,}000} = 0.0045\text{ mm}
        \]
        \item จาก $\mu\text{m}$ ไปเป็น $\text{nm}$ คือการเดินลงบันได 1 ขั้น (หน่วยใหญ่ไปหน่วยเล็ก) จึงนำมา\textbf{คูณด้วย $1{,}000$}
        \[
            L_{\text{nm}} = 4.5 \times 1{,}000 = 4{,}500\text{ nm}
        \]
    \end{itemize}
    \item \textbf{ขั้นที่ 2 แปลงปริมาตรสารละลาย $0.05\text{ mL}$ เป็น $\mu\text{L}$}
    \begin{itemize}
        \item จาก $\text{mL}$ ไปเป็น $\mu\text{L}$ คือการเดินลงบันได 1 ขั้น จึงนำมา\textbf{คูณด้วย $1{,}000$}
        \[
            \text{ปริมาตรไมโครปิเปต} = 0.05 \times 1{,}000 = 50\,\mu\text{L}
        \]
    \end{itemize}
\end{enumerate}
\textbf{คำตอบ}\quad แบคทีเรียมีความยาวเท่ากับ $0.0045\text{ mm}$ หรือ $4{,}500\text{ nm}$ และต้องตั้งค่าไมโครปิเปตที่ $50\,\mu\text{L}$
\end{worked}

\section{เลขนัยสำคัญ การปัดเลข การคำนวณทศนิยม ความคลาดเคลื่อน และคุณลักษณะเครื่องมือวัด}

\subsection{เลขนัยสำคัญ (Significant Figures) ในงานวัดทางจุลชีววิทยา}

ในการบันทึกผลการทดลองทางวิทยาศาสตร์ ตัวเลขทุกตัวที่เขียนลงในตารางบันทึกผลต้องสะท้อนความละเอียดของเครื่องมือวัดจริง ซึ่งเรียกว่า \textbf{เลขนัยสำคัญ} (\textbf{Significant Figures}) ประกอบด้วยตัวเลขที่อ่านได้แน่นอนจากขีดสเกล (\textit{Certain Digits}) รวมกับตัวเลขประมาณค่าตัวสุดท้ายอีก 1 ตำแหน่ง (\textit{Estimated Digit})

\begin{conceptbox}[กฎเกณฑ์สากล 5 ข้อในการนับจำนวนเลขนัยสำคัญ]
\begin{enumerate}
    \item \textbf{ตัวเลขที่ไม่ใช่ศูนย์ทุกตัว} นับเป็นเลขนัยสำคัญเสมอ เช่น $14.52\text{ g}$ (มี 4 ตัว), $3.8\text{ mm}$ (มี 2 ตัว)
    \item \textbf{เลขศูนย์ที่อยู่ระหว่างตัวเลขอื่น} นับเป็นเลขนัยสำคัญเสมอ เช่น $1.004\text{ mL}$ (มี 4 ตัว), $205\,\mu\text{m}$ (มี 3 ตัว)
    \item \textbf{เลขศูนย์ที่อยู่ด้านหน้า} \textbf{ไม่นับ}เป็นเลขนัยสำคัญ เพราะทำหน้าที่เพียงกำหนดตำแหน่งจุดทศนิยม เช่น $0.0025\text{ g}$ (มี 2 ตัว), $0.0408\text{ mL}$ (มี 3 ตัว)
    \item \textbf{เลขศูนย์ที่อยู่ด้านหลังและอยู่หลังจุดทศนิยม} \textbf{นับ}เป็นเลขนัยสำคัญ เพราะสะท้อนถึงความละเอียดของเครื่องมือ เช่น $1.5000\text{ g}$ จากเครื่องชั่งวิเคราะห์ 4 ตำแหน่ง (มี 5 ตัว), $10.0\text{ mL}$ (มี 3 ตัว)
    \item \textbf{เลขศูนย์ท้ายจำนวนเต็มที่ไม่มีจุดทศนิยม} มีความกำกวม เช่น $1{,}500\text{ CFU/mL}$ จึงต้องเขียนในรูป\textbf{สัญกรณ์วิทยาศาสตร์} ($a \times 10^b$) เสมอ เช่น $1.5 \times 10^3$ (มี 2 ตัว) หรือ $1.500 \times 10^3$ (มี 4 ตัว)
\end{enumerate}
\vspace{2pt}
\noindent\textbf{ข้อสังเกต}\quad \textbf{ตัวเลขแม่นตรง} (\textbf{Exact Numbers}) เช่น จำนวนจานเพาะเชื้อ 6 จาน หรือตัวคูณ $100\%$ ถือว่ามีเลขนัยสำคัญเป็นอนันต์ ($\infty$) จะไม่นำมาจำกัดเลขนัยสำคัญของผลลัพธ์
\end{conceptbox}

\subsection{หลักการปัดตัวเลขทางวิทยาศาสตร์ (Scientific Rounding Rules)}

เมื่อคำนวณแล้วได้ผลลัพธ์ที่มีจำนวนตัวเลขยาวเกินกว่าเลขนัยสำคัญที่ควรมี ต้องทำการปัดเศษตามเกณฑ์สากล:
\begin{itemize}
    \item \textbf{กรณีตัวเลขถัดไปน้อยกว่า 5}: ให้ \textbf{ปัดทิ้ง} (คงตัวเลขเดิมไว้) เช่น $4.163\text{ g}$ ปัดเหลือ 3 ตัวได้ $4.16\text{ g}$
    \item \textbf{กรณีตัวเลขถัดไปมากกว่า 5}: ให้ \textbf{ปัดขึ้น} (เพิ่มตัวเลขขึ้น 1) เช่น $4.167\text{ g}$ ปัดเหลือ 3 ตัวได้ $4.17\text{ g}$
    \item \textbf{กรณีตัวเลขถัดไปเท่ากับ 5 พอดี}: ให้ใช้\textbf{กฎปัดเข้าหาเลขคู่} (\textbf{Round-to-Even / Banker's Rounding}) เพื่อลดความเอนเอียงสะสม โดยถ้าเลขข้างหน้า 5 เป็น\textbf{เลขคู่ให้ปัดทิ้ง} และถ้าเป็น\textbf{เลขคี่ให้ปัดขึ้นเป็นเลขคู่} เช่น $2.450 \rightarrow 2.4$ และ $2.350 \rightarrow 2.4$
\end{itemize}

\subsection{การคำนวณทางคณิตศาสตร์ด้วยเลขนัยสำคัญและทศนิยม}

\begin{bioappbox}[กฎการคำนวณ 2 ประการที่นักจุลชีววิทยาต้องแยกแยะ]
\begin{enumerate}
    \item \textbf{การบวกและการลบ (Addition \& Subtraction):} ผลลัพธ์สุดท้ายต้องมี \textbf{จำนวนตำแหน่งทศนิยมเท่ากับตัวตั้งที่มีตำแหน่งทศนิยมน้อยที่สุด} โดยไม่ขึ้นกับจำนวนเลขนัยสำคัญรวม\\
    \textit{ตัวอย่าง}: ชั่งสารผสมในอาหารเลี้ยงเชื้อ: ผงวุ้น $15.2\text{ g}$ (ทศนิยม 1 ตำแหน่ง) + เปปโตน $1.245\text{ g}$ (3 ตำแหน่ง) + เกลือ $0.35\text{ g}$ (2 ตำแหน่ง) $= 16.795\text{ g} \rightarrow$ คำตอบที่ถูกต้องคือ \textbf{$16.8\text{ g}$} (ยึดทศนิยม 1 ตำแหน่งตาม $15.2\text{ g}$)
    \item \textbf{การคูณและการหาร (Multiplication \& Division):} ผลลัพธ์สุดท้ายต้องมี \textbf{จำนวนเลขนัยสำคัญเท่ากับตัวตั้งที่มีเลขนัยสำคัญน้อยที่สุด}\\
    \textit{ตัวอย่าง}: เตรียมสารละลายกลูโคสมวล $1.25\text{ g}$ (มี 3 ตัว) ในน้ำกลั่น $0.050\text{ L}$ (มี 2 ตัว):\\
    $C = \frac{1.25}{0.050} = 25\text{ g/L} \rightarrow$ คำตอบที่ถูกต้องคือ \textbf{$25\text{ g/L}$} (มีเลขนัยสำคัญ 2 ตัวตาม $0.050\text{ L}$)
\end{enumerate}
\vspace{2pt}
\noindent\textbf{ข้อควรระวัง}\quad ในการคำนวณหลายขั้นตอนต่อเนื่อง (\textit{Multi-step calculations}) ให้เก็บทศนิยมเผื่อไว้ 1--2 ตำแหน่ง (\textit{Guard digits}) ในขั้นตอนระหว่างทำ แล้วจึงปัดเศษตามกฎเลขนัยสำคัญใน\textbf{ขั้นตอนสุดท้ายเพียงครั้งเดียว}
\end{bioappbox}

\subsection{ความคลาดเคลื่อนในการวัด (Experimental Errors \& Uncertainty)}

ความคลาดเคลื่อนในการวัดจำแนกออกได้เป็น 3 ประเภทหลัก:
\begin{enumerate}
    \item \textbf{ความคลาดเคลื่อนเชิงระบบ (Systematic Error):} เบี่ยงเบนไปในทิศทางเดียวกันเสมอ เช่น ข้อบกพร่องของเครื่องมือวัด (\textit{Zero Error}) หรือไมโครปิเปตที่ลูกสูบสึกหรอ แก้ไขได้ด้วยการสอบเทียบ (\textit{Calibration}) และการกดปุ่มเซตศูนย์ (\textit{Tare})
    \item \textbf{ความคลาดเคลื่อนแบบสุ่ม (Random Error):} ผันแปรขึ้นลงไร้ทิศทางแน่นอน เช่น ความผันแปรของอุณหภูมิห้อง กระแสลมพัด หรือมุมมองสายตา (\textit{Parallax}) แก้ไขได้ด้วยการทำการทดลองซ้ำหลายครั้ง ($n \ge 3$) แล้วหาค่าเฉลี่ย ($\bar{x}$) และส่วนเบี่ยงเบนมาตรฐาน (\textit{SD})
    \item \textbf{ความคลาดเคลื่อนจากมนุษย์ (Gross / Human Error):} เกิดจากความประมาทเลินเล่อ เช่น ดูดสารผิดจังหวะ หรืออ่านสเกลผิดขีด ต้องตัดข้อมูลชุดนั้นทิ้งแล้วทดลองใหม่
\end{enumerate}

\deepskinfig{figures/sec_3d/ch01_accuracy_precision_3d.jpg}{แบบจำลอง 3 มิติเปรียบเทียบความแตกต่างระหว่างความแม่น (Accuracy) และความเที่ยง (Precision) ผ่านเป้ายิง 4 รูปแบบ: (1) แม่นสูง-เที่ยงสูง เกาะกลุ่มตรงกลางเป้า (2) แม่นต่ำ-เที่ยงสูง เกาะกลุ่มแน่นแต่นอกเป้า (3) แม่นสูง-เที่ยงต่ำ กระจายรอบจุดกึ่งกลาง (4) แม่นต่ำ-เที่ยงต่ำ กระจายไร้ทิศทาง}

\deepskinfig{figures/sec_3d/ch01_micropipette_3d.jpg}{แบบจำลอง 3 มิติเชิงเทคนิคแสดงภาพตัดขวางของไมโครปิเปต P1000 กลไกปุ่มกด 2 จังหวะ (First Stop สำหรับดูดสาร และ Second Stop สำหรับเป่าสารออก) พร้อมสปริงสอบเทียบและหน้าปัดดิจิทัล $1{,}000\,\mu\text{L}$}

หากเราต้องการตรวจสอบว่าไมโครปิเปตในห้องปฏิบัติการยังมีความแม่นยำหรือไม่ เราสามารถใช้วิธีชั่งมวลของน้ำกลั่นที่ดูดออกมา เนื่องจากน้ำบริสุทธิ์ที่อุณหภูมิห้องมีความหนาแน่นประมาณ $1\text{ g/mL}$ หรือกล่าวได้ว่า \textbf{น้ำกลั่นปริมาตร $1{,}000\,\mu\text{L}$ ($1\text{ mL}$) จะมีมวลเท่ากับ $1.0000\text{ g}$ และน้ำกลั่น $100\,\mu\text{L}$ จะมีมวลเท่ากับ $0.1000\text{ g}$} โดยคำนวณเปอร์เซ็นต์ความคลาดเคลื่อนสัมพัทธ์ได้ดังนี้
\begin{equation*}
    \% \text{Relative Error} = \frac{|x_{\text{measured}} - x_{\text{true}}|}{x_{\text{true}}} \times 100\%
\end{equation*}
โดยที่ $x_{\text{measured}}$ คือค่าที่วัดได้จริงจากเครื่องมือในห้องปฏิบัติการ และ $x_{\text{true}}$ คือค่ามาตรฐานที่แท้จริง

\begin{warningbox}[ห้ามกดปุ่มไมโครปิเปตเลยจังหวะแรกขณะดูดสารละลาย]
ไมโครปิเปตมีปุ่มกด 2 จังหวะ (\textbf{First Stop} และ \textbf{Second Stop}) ขณะจุ่มปลายทิปเพื่อดูดสารละลายหรือเชื้อจุลินทรีย์ \textbf{ต้องกดลงไปเพียงจังหวะแรกเท่านั้น} แล้วค่อยๆ ผ่อนนิ้วขึ้น หากเผลอกดลงไปถึงจังหวะที่สองขณะดูดสาร จะทำให้ได้ปริมาตรเกินกว่าที่ตั้งไว้ถึง 10--15\% ส่วนจังหวะที่สองมีไว้ใช้ตอนปล่อยสารออกจากปลายทิปให้หมดจดเท่านั้น
\end{warningbox}

\subsection{คุณลักษณะสำคัญของเครื่องมือวัด: ความแม่น ความเที่ยง ความละเอียด และความไว}

ในการเลือกใช้เครื่องมือวัดสำหรับงานวิจัยจุลชีววิทยา ต้องเข้าใจ 4 มิติสำคัญทางมาตรวิทยา:
\begin{itemize}
    \item \textbf{ความแม่น (Accuracy):} ความใกล้เคียงระหว่างค่าที่วัดได้จริงกับค่าจริงมาตรฐาน ประเมินจาก $\% \text{Relative Error}$
    \item \textbf{ความเที่ยง (Precision):} ความสามารถในการวัดซ้ำแล้วได้ค่าเดิมสม่ำเสมอ ประเมินจากส่วนเบี่ยงเบนมาตรฐาน ($SD$) หรือ $\%CV$
    \item \textbf{ความละเอียด (Resolution / Least Count):} ขนาดการเปลี่ยนแปลงที่เล็กที่สุดของปริมาณที่วัดซึ่งเครื่องมือสามารถตรวจวัดและแสดงผลได้ เช่น ไมโครมิเตอร์วัดได้ละเอียด $0.01\text{ mm}$ และเครื่องชั่งวิเคราะห์ 4 ตำแหน่งวัดได้ละเอียด $0.0001\text{ g}$
    \item \textbf{ความไว (Sensitivity):} อัตราส่วนระหว่างการเปลี่ยนแปลงสัญญาณตอบสนองต่อการเปลี่ยนแปลงปริมาณที่วัด $S = \frac{\Delta \text{Output}}{\Delta \text{Input}} = \frac{\Delta y}{\Delta x}$ เช่น ความชันของกราฟมาตรฐานการดูดกลืนแสงในเครื่องสเปกโทรโฟโตมิเตอร์
\end{itemize}

\noindent\textbf{ผลลัพธ์การเรียนรู้ที่ประเมิน}\quad CLO1 (ความรู้การวัดและแปลงหน่วย), CLO3 (วินัย ความปลอดภัย และความซื่อสัตย์ในการบันทึกข้อมูลจริง), CLO4 (การทำงานร่วมกันและเขียนรายงาน)

\vspace{4pt}
\noindent\textbf{1. วัสดุและอุปกรณ์ (ดังแสดงใน\mbox{ภาพที่} 1.5)}
\begin{itemize}
    \item เวอร์เนียร์คาลิปเปอร์ (ความละเอียด $0.05\text{ mm}$) และไมโครมิเตอร์ (ความละเอียด $0.01\text{ mm}$)
    \item จานเพาะเชื้อแก้ว (Petri Dish), หลอดทดลอง (Test Tube) และแผ่นสไลด์กล้องจุลทรรศน์
    \item ไมโครปิเปตขนาด P1000 ($100\text{--}1{,}000\,\mu\text{L}$) และ P200 ($20\text{--}200\,\mu\text{L}$) พร้อมทิปปลอดเชื้อ
    \item เครื่องชั่งดิจิทัลละเอียดทศนิยม 4 ตำแหน่ง บีกเกอร์ใส่น้ำกลั่น และผงวุ้นอาหารเลี้ยงเชื้อจำลอง (โซเดียมคลอไรด์)
\end{itemize}

\noindent\textbf{2. ขั้นตอนการทดลอง}
\begin{enumerate}
    \item \textbf{ตอนที่ 1 การวัดมิติภาชนะทางจุลชีววิทยา}\quad ใช้เวอร์เนียร์คาลิปเปอร์วัดเส้นผ่านศูนย์กลางภายในและความลึกของจานเพาะเชื้อจำนวน 3 ครั้ง และใช้ไมโครมิเตอร์วัดความหนาของแผ่นสไลด์แก้วจำนวน 3 ครั้ง บันทึกค่าเฉลี่ยและแปลงหน่วยเป็นไมโครเมตร ($\mu\text{m}$)
    \item \textbf{ตอนที่ 2 การตรวจสอบความแม่นและความเที่ยงของไมโครปิเปต}\quad วางบีกเกอร์พลาสติกขนาดเล็กบนเครื่องชั่งดิจิทัลแล้วกดปุ่มเซตศูนย์ (\textbf{Tare}) จากนั้นตั้งค่าไมโครปิเปต P1000 ที่ปริมาตร $500\,\mu\text{L}$ ดูดน้ำกลั่นหยดลงในบีกเกอร์ อ่านค่ามวลเป็นกรัม ($\text{g}$) ทำซ้ำ 3 ครั้ง แล้วคำนวณค่าเฉลี่ยและเปอร์เซ็นต์ความคลาดเคลื่อนเทียบกับค่ามาตรฐาน $0.5000\text{ g}$
\end{enumerate}

\section*{คำถามท้ายบทที่ 1 (ทบทวนแนวคิดและการคำนวณเบื้องต้น)}
\begin{enumerate}
    \item เซลล์ยีสต์ \textit{Saccharomyces cerevisiae} มีเส้นผ่านศูนย์กลางประมาณ $6\,\mu\text{m}$ จงแปลงค่าดังกล่าวให้เป็นหน่วยมิลลิเมตร ($\text{mm}$) และนาโนเมตร ($\text{nm}$) โดยใช้เทคนิคบันไดขั้นละ $1{,}000$
    \item หากต้องการเตรียมสารละลายยาปฏิชีวนะปริมาตร $0.25\text{ mL}$ นักศึกษาควรเลือกใช้ไมโครปิเปตขนาดใดระหว่าง P20, P200 หรือ P1000 และต้องตั้งตัวเลขเป็นกี่ไมโครลิตร ($\mu\text{L}$) เพราะเหตุใด
    \item ในการทดสอบไมโครปิเปตที่ตั้งค่าไว้ $200\,\mu\text{L}$ นักศึกษาชั่งมวลน้ำกลั่น 3 ครั้ง ได้ค่า $0.198\text{ g}$, $0.199\text{ g}$ และ $0.197\text{ g}$ จงหาค่าเฉลี่ยและเปอร์เซ็นต์ความคลาดเคลื่อนเทียบกับค่ามาตรฐาน $0.200\text{ g}$
\end{enumerate}
